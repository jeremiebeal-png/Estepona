/* ═══════════════════════════════════════════════════════════════════════════
 * ENVOI D'EMAILS VIA MICROSOFT 365 (Microsoft Graph API)
 * ---------------------------------------------------------------------------
 * Remplace Resend. Envoie depuis support@dsihotel.fr (boîte M365, domaine
 * déjà vérifié dans le tenant) avec un Reply-To forcé sur support@dsihotel.com
 * afin que les réponses des clients arrivent sur Cloudflare Email Routing

 *
 * VARIABLES CLOUDFLARE À CONFIGURER (Settings → Variables) :
 *   - MS_TENANT_ID       (Texte)   : ID du tenant Entra ID (Azure AD)
 *   - MS_CLIENT_ID       (Texte)   : Application (client) ID de l'App Registration
 *   - MS_CLIENT_SECRET   (Secret)  : valeur du client secret créé dans l'App
 *   - MS_SENDER          (Texte, optionnel) : boîte expéditrice.
 *       Défaut : support@dsihotel.fr
 *   - MS_REPLY_TO        (Texte, optionnel) : adresse de réponse forcée.
 *       Défaut : support@dsihotel.com
 *
 * PERMISSION AZURE REQUISE : Mail.Send (type Application) + consentement admin.
 * Restreindre de préférence via ApplicationAccessPolicy (PowerShell) à la
 * seule boîte support@dsihotel.fr.
 * ═══════════════════════════════════════════════════════════════════════════ */

const MS_SENDER_DEFAUT = "support@dsihotel.fr";
const MS_REPLY_TO_DEFAUT = "support@dsihotel.com";
const PLANNING_MAILBOX_DEFAUT = "info@dsihotel.fr"; // boîte dont le calendrier sert de planning d'équipe

// Cache du token en mémoire du Worker (isolate). Évite de redemander un token
// à chaque envoi. Le token M365 dure ~3600 s ; on garde une marge de sécurité.
let _msTokenCache = { token: null, exp: 0 };

async function msGetToken(env) {
  const now = Math.floor(Date.now() / 1000);
  if (_msTokenCache.token && _msTokenCache.exp > now + 60) {
    return _msTokenCache.token;
  }
  const tenant = env.MS_TENANT_ID;
  const url = `https://login.microsoftonline.com/${tenant}/oauth2/v2.0/token`;
  const params = new URLSearchParams({
    client_id: env.MS_CLIENT_ID,
    client_secret: env.MS_CLIENT_SECRET,
    scope: "https://graph.microsoft.com/.default",
    grant_type: "client_credentials",
  });
  const res = await fetch(url, {
    method: "POST",
    headers: { "Content-Type": "application/x-www-form-urlencoded" },
    body: params.toString(),
  });
  if (!res.ok) {
    const t = await res.text().catch(() => "");
    throw new Error("Token M365 refusé (" + res.status + ") " + t.slice(0, 200));
  }
  const data = await res.json();
  _msTokenCache = {
    token: data.access_token,
    exp: now + (data.expires_in || 3600),
  };
  return data.access_token;
}

// Normalise une liste d'adresses (chaîne, tableau, séparateurs , ou ;) vers
// le format Graph : [{ emailAddress: { address } }].
function msRecipients(value) {
  if (!value) return [];
  const arr = Array.isArray(value) ? value : String(value).split(/[,;]/);
  return arr
    .map((a) => String(a).trim())
    .filter((a) => /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(a))
    .map((a) => ({ emailAddress: { address: a } }));
}

/**
 * Envoie un email via Microsoft Graph.
 * @param {object} env
 * @param {object} msg
 *   - to          (string | string[])           destinataires (requis)
 *   - subject     (string)                        objet (requis)
 *   - text        (string)                        corps texte (fallback si pas de html)
 *   - html        (string, optionnel)             corps HTML
 *   - cc          (string | string[], optionnel)  copies
 *   - replyTo     (string, optionnel)             remplace le Reply-To par défaut
 *   - fromName    (string, optionnel)             nom affiché de l'expéditeur
 *   - attachments ([{ filename, content(base64) }], optionnel)
 * @returns {Promise<{ok:boolean, status:number, error?:string}>}
 */
async function sendViaGraph(env, msg) {
  const sender = (env.MS_SENDER || MS_SENDER_DEFAUT).trim();
  const replyToAddr = (msg.replyTo || env.MS_REPLY_TO || MS_REPLY_TO_DEFAUT).trim();

  const toList = msRecipients(msg.to);
  if (toList.length === 0) {
    return { ok: false, status: 400, error: "Aucun destinataire valide" };
  }

  const useHtml = !!msg.html;
  const body = {
    contentType: useHtml ? "HTML" : "Text",
    content: useHtml ? msg.html : (msg.text || ""),
  };

  const message = {
    subject: msg.subject || "(Sans objet)",
    body,
    toRecipients: toList,
  };

  const ccList = msRecipients(msg.cc);
  if (ccList.length > 0) message.ccRecipients = ccList;

  const replyList = msRecipients(replyToAddr);
  if (replyList.length > 0) message.replyTo = replyList;

  // Nom affiché de l'expéditeur (facultatif). L'adresse reste la boîte M365.
  const cleanName = String(msg.fromName || "").replace(/["<>\r\n]/g, "").trim();
  if (cleanName) {
    message.from = { emailAddress: { name: cleanName, address: sender } };
  }

  // Pièces jointes : Graph attend @odata.type + contentBytes (base64).
  // Si l'attachement porte un "cid" (Content-ID), on le marque inline : il est
  // alors référencé dans le HTML par <img src="cid:LE_CID"> et s'affiche DANS le
  // corps du mail (façon Outlook) au lieu d'apparaître en pièce jointe séparée.
  if (Array.isArray(msg.attachments) && msg.attachments.length > 0) {
    message.attachments = msg.attachments
      .filter((a) => a && a.content)
      .map((a) => {
        const att = {
          "@odata.type": "#microsoft.graph.fileAttachment",
          name: a.filename || "piece-jointe",
          contentBytes: a.content,
        };
        if (a.contentType) att.contentType = a.contentType;
        if (a.cid) {
          att.contentId = String(a.cid);
          att.isInline = true;
        }
        return att;
      });
  }

  let token;
  try {
    token = await msGetToken(env);
  } catch (e) {
    return { ok: false, status: 500, error: String((e && e.message) || e) };
  }

  const endpoint =
    `https://graph.microsoft.com/v1.0/users/${encodeURIComponent(sender)}/sendMail`;
  const res = await fetch(endpoint, {
    method: "POST",
    headers: {
      Authorization: `Bearer ${token}`,
      "Content-Type": "application/json",
    },
    body: JSON.stringify({ message, saveToSentItems: true }),
  });

  // Graph renvoie 202 Accepted sans corps en cas de succès.
  if (res.status === 202) return { ok: true, status: 202 };

  const detail = await res.text().catch(() => "");
  return { ok: false, status: res.status, error: detail.slice(0, 300) };
}

function corsHeaders() {
  const allowed = ALLOWED_ORIGINS.includes(CURRENT_ORIGIN) ? CURRENT_ORIGIN : ALLOWED_ORIGINS[0];
  return {
    "Access-Control-Allow-Origin": allowed,
    "Access-Control-Allow-Methods": "POST, OPTIONS",
    "Access-Control-Allow-Headers": "Content-Type, Authorization",
  };
}

function json(body, status = 200) {
  return new Response(JSON.stringify(body), { status, headers: { "Content-Type": "application/json", ...corsHeaders() } });
}

function sbHeaders(env) {
  return {
    apikey: env.SUPABASE_SERVICE_ROLE_KEY,
    ...(String(env.SUPABASE_SERVICE_ROLE_KEY || "").startsWith("sb_") ? {} : { Authorization: `Bearer ${env.SUPABASE_SERVICE_ROLE_KEY}` }),
  };
}

function sbJsonHeaders(env) {
  return { ...sbHeaders(env), "Content-Type": "application/json" };
}

// ===================== TOTP "maison" (RFC 6238) =====================
// Implémentation sans dépendance, via WebCrypto (disponible sur Workers).
const TOTP_B32 = "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";

// Génère un secret base32 aléatoire (par défaut 20 octets -> 32 caractères).
function genTotpSecret(bytes = 20) {
  const buf = new Uint8Array(bytes);
  crypto.getRandomValues(buf);
  let bits = 0, value = 0, out = "";
  for (let i = 0; i < buf.length; i++) {
    value = (value << 8) | buf[i];
    bits += 8;
    while (bits >= 5) {
      out += TOTP_B32[(value >>> (bits - 5)) & 31];
      bits -= 5;
    }
  }
  if (bits > 0) out += TOTP_B32[(value << (5 - bits)) & 31];
  return out;
}

// Décode un secret base32 en octets.
function b32Decode(secret) {
  const clean = String(secret || "").toUpperCase().replace(/=+$/g, "").replace(/\s/g, "");
  let bits = 0, value = 0;
  const out = [];
  for (const ch of clean) {
    const idx = TOTP_B32.indexOf(ch);
    if (idx === -1) continue;
    value = (value << 5) | idx;
    bits += 5;
    if (bits >= 8) {
      out.push((value >>> (bits - 8)) & 0xff);
      bits -= 8;
    }
  }
  return new Uint8Array(out);
}

// Calcule le code TOTP (6 chiffres) pour un secret et un "compteur" de fenêtre.
async function totpAt(secret, counter) {
  const key = b32Decode(secret);
  const msg = new Uint8Array(8);
  let c = counter;
  for (let i = 7; i >= 0; i--) { msg[i] = c & 0xff; c = Math.floor(c / 256); }
  const cryptoKey = await crypto.subtle.importKey(
    "raw", key, { name: "HMAC", hash: "SHA-1" }, false, ["sign"]
  );
  const sig = new Uint8Array(await crypto.subtle.sign("HMAC", cryptoKey, msg));
  const offset = sig[sig.length - 1] & 0x0f;
  const bin = ((sig[offset] & 0x7f) << 24) | ((sig[offset + 1] & 0xff) << 16) |
              ((sig[offset + 2] & 0xff) << 8) | (sig[offset + 3] & 0xff);
  return String(bin % 1000000).padStart(6, "0");
}

// Vérifie un code sur une fenêtre de +/- 1 pas (30 s) pour tolérer le décalage d'horloge.
async function totpVerify(secret, code, stepSeconds = 30, window = 1) {
  const clean = String(code || "").replace(/\s/g, "");
  if (!/^[0-9]{6}$/.test(clean)) return false;
  const counter = Math.floor(Date.now() / 1000 / stepSeconds);
  for (let w = -window; w <= window; w++) {
    if (await totpAt(secret, counter + w) === clean) return true;
  }
  return false;
}

// Construit l'URI otpauth:// (pour le QR code).
function otpauthUri(secret, label, issuer) {
  const l = encodeURIComponent(label);
  const i = encodeURIComponent(issuer);
  return `otpauth://totp/${i}:${l}?secret=${secret}&issuer=${i}&algorithm=SHA1&digits=6&period=30`;
}

// Récupère l'hôtel (id + emails) à partir de l'email de connexion fourni.
// Génère un PDF d'attestation minimal (sans dépendance) et le renvoie en base64.
function buildAttestationPdfBase64(lignes) {
  const esc = (s) => String(s == null ? "" : s).replace(/\\/g, "\\\\").replace(/\(/g, "\\(").replace(/\)/g, "\\)");
  let content = "BT\n";
  // bandeau (rectangle bleu foncé)
  let pre = "0.10 0.10 0.24 rg\n0 800 595 42 re\nf\n";
  for (const ln of lignes) {
    const size = ln.size || 11;
    const font = ln.bold ? "F2" : "F1";
    const col = ln.white ? "1 1 1 rg" : (ln.gold ? "0.66 0.53 0.18 rg" : "0.15 0.15 0.20 rg");
    content += `${col}\n/${font} ${size} Tf\n1 0 0 1 ${ln.x || 55} ${ln.y} Tm\n(${esc(ln.text)}) Tj\n`;
  }
  content += "ET";
  const stream = pre + content;
  const objs = [];
  objs.push("<< /Type /Catalog /Pages 2 0 R >>");
  objs.push("<< /Type /Pages /Kids [3 0 R] /Count 1 >>");
  objs.push("<< /Type /Page /Parent 2 0 R /MediaBox [0 0 595 842] /Resources << /Font << /F1 5 0 R /F2 6 0 R >> >> /Contents 4 0 R >>");
  objs.push(`<< /Length ${stream.length} >>\nstream\n${stream}\nendstream`);
  objs.push("<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica >>");
  objs.push("<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica-Bold >>");
  let pdf = "%PDF-1.4\n";
  const offsets = [];
  objs.forEach((o, i) => { offsets.push(pdf.length); pdf += `${i + 1} 0 obj\n${o}\nendobj\n`; });
  const xref = pdf.length;
  pdf += `xref\n0 ${objs.length + 1}\n0000000000 65535 f \n`;
  offsets.forEach(off => { pdf += String(off).padStart(10, "0") + " 00000 n \n"; });
  pdf += `trailer\n<< /Size ${objs.length + 1} /Root 1 0 R >>\nstartxref\n${xref}\n%%EOF`;
  // Encode en base64 (latin1)
  let bin = "";
  for (let i = 0; i < pdf.length; i++) bin += String.fromCharCode(pdf.charCodeAt(i) & 0xff);
  return btoa(bin);
}

function extractEmail(raw) {
  if (!raw) return "";
  const m = String(raw).match(/<([^>]+)>/);
  if (m) return m[1].trim().toLowerCase();
  const m2 = String(raw).match(/[\w.+-]+@[\w.-]+\.\w+/);
  return m2 ? m2[0].trim().toLowerCase() : "";
}

function extractName(raw) {
  if (!raw) return "";
  const m = String(raw).match(/^\s*"?([^"<]+?)"?\s*</);
  return m ? m[1].trim() : "";
}

function stripHtml(html) {
  if (!html) return "";
  return String(html)
    .replace(/<style[\s\S]*?<\/style>/gi, "")
    .replace(/<script[\s\S]*?<\/script>/gi, "")
    .replace(/<[^>]+>/g, " ")
    .replace(/&nbsp;/g, " ")
    .replace(/\s+/g, " ")
    .trim();
}

// Recherche fiable d'un utilisateur Auth par son email EXACT.
// Le paramètre ?email= de GoTrue (Supabase Auth) n'est pas fiable selon la
// version (souvent ignoré -> renvoie tous les users). On pagine donc la liste
// et on compare l'email nous-mêmes. Retourne l'objet user, ou null.
async function trouverUserParEmail(env, mail) {
  const cible = String(mail || "").trim().toLowerCase();
  if (!cible) return null;
  const PER_PAGE = 200;
  for (let page = 1; page <= 50; page++) {
    const res = await fetch(
      `${env.SUPABASE_URL}/auth/v1/admin/users?page=${page}&per_page=${PER_PAGE}`,
      { headers: sbHeaders(env) }
    );
    if (!res.ok) return null;
    const data = await res.json().catch(() => ({}));
    const users = data.users || [];
    if (users.length === 0) break; // vraie fin de liste (Supabase peut plafonner per_page)
    const trouve = users.find(u => String(u.email || "").trim().toLowerCase() === cible);
    if (trouve) return trouve;
  }
  return null;
}

// Retrouve le compte Auth d'un hôtel pour un niveau donné (supervisor / client)
// via son profil (profiles.hotel_id + role) : ne dépend pas de l'email.
// Repli sur la recherche par email si aucun profil n'est trouvé.
async function trouverUserDuCompteHotel(env, hotelId, level, email) {
  const cible = String(email || "").trim().toLowerCase();
  const pRes = await fetch(
    `${env.SUPABASE_URL}/rest/v1/profiles?hotel_id=eq.${hotelId}&role=eq.${level}&select=id`,
    { headers: sbHeaders(env) }
  );
  const profs = await pRes.json().catch(() => []);
  const liste = Array.isArray(profs) ? profs : [];
  for (const p of liste) {
    const uRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users/${p.id}`, { headers: sbHeaders(env) });
    if (!uRes.ok) continue;
    const u = await uRes.json().catch(() => null);
    if (!u || !u.id) continue;
    if (liste.length === 1 || String(u.email || "").trim().toLowerCase() === cible) return u;
  }
  return await trouverUserParEmail(env, cible);
}

// Adresse email valide pour Supabase Auth : pas d'accents, d'espaces ni de caractères spéciaux.
function emailValide(mail) {
  return /^[a-z0-9._%+\-]+@[a-z0-9.\-]+\.[a-z]{2,}$/.test(String(mail || ""));
}

// Session + profil (role, hotel_id) de l'appelant. Retourne null si non connecté.
async function sessionEtProfil(request, env) {
  const user = await verifySession(request, env);
  if (!user || !user.id) return null;
  const r = await fetch(`${env.SUPABASE_URL}/rest/v1/profiles?id=eq.${user.id}&select=role,hotel_id`, { headers: sbHeaders(env) });
  const arr = await r.json().catch(() => []);
  const prof = (Array.isArray(arr) && arr[0]) || {};
  return { user, role: prof.role || null, hotelId: prof.hotel_id || null };
}

async function deleteAuthUserByEmail(env, mail) {
  if (!mail) return;
  const targetUser = await trouverUserParEmail(env, mail);
  if (!targetUser) return;
  await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users/${targetUser.id}`, {
    method: "DELETE",
    headers: sbHeaders(env),
  });
}

async function verifySession(request, env) {
  const authHeader = request.headers.get("Authorization") || "";
  const token = authHeader.replace("Bearer ", "");
  if (!token) return null;
  const userRes = await fetch(`${env.SUPABASE_URL}/auth/v1/user`, {
    headers: { Authorization: `Bearer ${token}`, apikey: env.SUPABASE_SERVICE_ROLE_KEY },
  });
  if (!userRes.ok) return null;
  return await userRes.json();
}

async function verifyAdmin(request, env) {
  const authHeader = request.headers.get("Authorization") || "";
  const token = authHeader.replace("Bearer ", "");
  if (!token) return null;

  const userRes = await fetch(`${env.SUPABASE_URL}/auth/v1/user`, {
    headers: { Authorization: `Bearer ${token}`, apikey: env.SUPABASE_SERVICE_ROLE_KEY },
  });
  if (!userRes.ok) return null;
  const user = await userRes.json();

  const profRes = await fetch(`${env.SUPABASE_URL}/rest/v1/profiles?id=eq.${user.id}&select=role`, {
    headers: sbHeaders(env),
  });
  if (!profRes.ok) return null;
  const profiles = await profRes.json();
  if (!profiles[0] || profiles[0].role !== "admin") return null;

  return user;
}

async function verifySuperAdmin(request, env) {
  const authHeader = request.headers.get("Authorization") || "";
  const token = authHeader.replace("Bearer ", "");
  if (!token) return null;

  const userRes = await fetch(`${env.SUPABASE_URL}/auth/v1/user`, {
    headers: { Authorization: `Bearer ${token}`, apikey: env.SUPABASE_SERVICE_ROLE_KEY },
  });
  if (!userRes.ok) return null;
  const user = await userRes.json();

  if ((user.email || "").toLowerCase() !== SUPERADMIN_EMAIL) return null;
  return user;
}

Description :

          if (level !== "supervisor" && level !== "client") return json({ error: "Niveau invalide" }, 400);

          // Récupère l'hôtel et vérifie qu'un compte de ce niveau n'existe pas déjà
          const hRes = await fetch(`${env.SUPABASE_URL}/rest/v1/hotels?id=eq.${hotelId}&select=id,account_email,standard_email`, {
            headers: sbHeaders(env),
          });
          const hArr = await hRes.json().catch(() => []);
          if (!hArr[0]) return json({ error: "Hôtel introuvable" }, 404);
          const hotel = hArr[0];

          if (level === "supervisor" && hotel.account_email) {
            return json({ error: "Cet hôtel a déjà un compte administrateur." }, 400);
          }
          if (level === "client" && hotel.standard_email) {
            return json({ error: "Cet hôtel a déjà un compte standard." }, 400);
          }

          // Crée le compte auth
          const uRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users`, {
            method: "POST",
            headers: sbJsonHeaders(env),
            body: JSON.stringify({ email, password, email_confirm: true }),
          });
          const uData = await uRes.json();
          if (!uRes.ok) return json({ error: "Erreur création compte", details: uData }, uRes.status);

          // Crée le profil lié à l'hôtel
          const pRes = await fetch(`${env.SUPABASE_URL}/rest/v1/profiles`, {
            method: "POST",
            headers: sbJsonHeaders(env),
            body: JSON.stringify({ id: uData.id, role: level, hotel_id: hotelId }),
          });
          if (!pRes.ok) {
            await deleteAuthUserByEmail(env, email);
            const e = await pRes.json().catch(() => ({}));
            return json({ error: "Erreur création profil", details: e }, 500);
          }

          // Met à jour l'email correspondant sur la fiche hôtel
          const patchField = level === "supervisor" ? { account_email: email } : { standard_email: email };
          await fetch(`${env.SUPABASE_URL}/rest/v1/hotels?id=eq.${hotelId}`, {
            method: "PATCH",
            headers: sbJsonHeaders(env),
            body: JSON.stringify(patchField),
          });

          return json({ success: true, email, level });
        }

        // --- Modifier l'adresse email (et éventuellement le mot de passe) d'un compte ---
        // Le portail ne permettait pas de changer l'email d'un compte existant
        // (Auth impose 1 seul ADMIN et 1 seul STD par hôtel). Cette route met à jour
        // l'email côté Supabase Auth ET le champ correspondant de la fiche hôtel,
        // sans supprimer le compte (l'ID, le profil et les tickets sont conservés).
        if (url.pathname === "/admin/update-account-email") {
          const { hotelId, level, oldEmail, newEmail, newPassword } = body;
          if (!hotelId || !level || !oldEmail || !newEmail) {
            return json({ error: "Champs manquants (hotelId, level, oldEmail, newEmail)" }, 400);
          }
          if (level !== "supervisor" && level !== "client") return json({ error: "Niveau invalide" }, 400);

          const emailNorm = String(newEmail).trim().toLowerCase();
          const ancienNorm = String(oldEmail).trim().toLowerCase();
          if (!emailValide(emailNorm)) {
            return json({ error: "Adresse invalide : pas d'accents, d'espaces ni de caractères spéciaux (ex. cezanne_admin@dsihotel.fr)." }, 400);
          }

          // 1) Vérifie que l'hôtel existe et que le compte de ce niveau correspond bien à oldEmail.
          const hRes = await fetch(`${env.SUPABASE_URL}/rest/v1/hotels?id=eq.${hotelId}&select=id,account_email,standard_email`, {
            headers: sbHeaders(env),
          });
          const hArr = await hRes.json().catch(() => []);
          if (!hArr[0]) return json({ error: "Hôtel introuvable" }, 404);
          const hotel = hArr[0];
          const emailActuel = level === "supervisor" ? hotel.account_email : hotel.standard_email;
          if (!emailActuel) {
            return json({ error: "Aucun compte de ce niveau sur cet hôtel." }, 400);
          }
          if (String(emailActuel).trim().toLowerCase() !== ancienNorm) {
            return json({ error: "L'ancienne adresse fournie ne correspond pas au compte enregistré." }, 400);
          }

          // 2) Si l'adresse ne change pas, on peut quand même mettre à jour le mot de passe.
          //    Sinon, on vérifie que la nouvelle adresse n'est pas déjà utilisée par un autre compte.
          if (emailNorm !== ancienNorm) {
            // NB : le paramètre ?email= de GoTrue (Supabase Auth) n'est pas fiable
            // (souvent ignoré -> renvoie tous les users). On filtre donc nous-mêmes
            // sur l'adresse exacte, via l'helper trouverUserParEmail.
            const existant = await trouverUserParEmail(env, emailNorm);
            if (existant) {
              return json({ error: "Cette adresse email est déjà utilisée par un autre compte." }, 400);
            }
          }

          // 3) Retrouve le compte auth via le profil de l'hôtel (fiable), sinon par l'ancienne adresse.
          const targetUser = await trouverUserDuCompteHotel(env, hotelId, level, ancienNorm);
          if (!targetUser) return json({ error: "Compte introuvable dans Auth pour l'ancienne adresse." }, 404);

          // 4) Met à jour l'email (+ mot de passe si fourni) côté Supabase Auth.
          const updatePayload = { email: emailNorm, email_confirm: true };
          if (newPassword) updatePayload.password = newPassword;
          const updRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users/${targetUser.id}`, {
            method: "PUT",
            headers: sbJsonHeaders(env),
            body: JSON.stringify(updatePayload),
          });
          if (!updRes.ok) {
            const updErr = await updRes.json().catch(() => ({}));
            const raison = updErr.msg || updErr.message || updErr.error_description || updErr.error || ("statut " + updRes.status);
            return json({ error: "Erreur mise à jour du compte Auth : " + raison, details: updErr }, 500);
          }

          // 5) Met à jour l'email correspondant sur la fiche hôtel (cohérence portail + rattachement tickets).
          const patchField = level === "supervisor" ? { account_email: emailNorm } : { standard_email: emailNorm };
          const patchRes = await fetch(`${env.SUPABASE_URL}/rest/v1/hotels?id=eq.${hotelId}`, {
            method: "PATCH",
            headers: sbJsonHeaders(env),
            body: JSON.stringify(patchField),
          });
          if (!patchRes.ok) {
            const pErr = await patchRes.text().catch(() => "");
            return json({ error: "Compte modifié mais mise à jour de la fiche hôtel échouée", detail: pErr.slice(0, 200) }, 500);
          }

          return json({ success: true, email: emailNorm, level, passwordChanged: !!newPassword });
        }

        // --- Réinitialiser un mot de passe ---
        if (url.pathname === "/admin/reset-password") {
          const { userEmail, newPassword } = body;
          if (!userEmail || !newPassword) return json({ error: "Champs manquants" }, 400);

          // Recherche sur l'adresse EXACTE (l'ancien ?email= pouvait renvoyer un autre compte).
          const targetUser = await trouverUserParEmail(env, userEmail);
          if (!targetUser) return json({ error: "Utilisateur introuvable" }, 404);

          const updateRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users/${targetUser.id}`, {
            method: "PUT",
            headers: sbJsonHeaders(env),
            body: JSON.stringify({ password: newPassword }),
          });
          if (!updateRes.ok) {
            const updErrData = await updateRes.json().catch(() => ({}));
            return json({ error: "Erreur réinitialisation", details: updErrData }, 500);
          }
          return json({ success: true });
        }

        // --- Supprimer un client ---
        if (url.pathname === "/admin/delete-client") {
          const { hotelId, userEmail, standardEmail } = body;
          if (!hotelId) return json({ error: "Champ manquant : hotelId" }, 400);

          const delHotelRes = await fetch(`${env.SUPABASE_URL}/rest/v1/hotels?id=eq.${hotelId}`, {
            method: "DELETE",
            headers: sbHeaders(env),
          });
          if (!delHotelRes.ok) return json({ error: "Erreur suppression hôtel" }, 500);

          for (const mail of [userEmail, standardEmail].filter(Boolean)) {
            await deleteAuthUserByEmail(env, mail);
          }

          return json({ success: true });
        }

        // --- Lister les admins (super-admin uniquement) ---
        if (url.pathname === "/admin/list-admins") {
          if (!(await verifySuperAdmin(request, env))) return json({ error: "Réservé au super-admin" }, 403);

          const profRes = await fetch(
            `${env.SUPABASE_URL}/rest/v1/profiles?role=eq.admin&select=id,permissions`,
            { headers: sbHeaders(env) }
          );
          const profs = await profRes.json();

          const admins = [];
          for (const p of profs) {
            const uRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users/${p.id}`, { headers: sbHeaders(env) });
            const u = uRes.ok ? await uRes.json() : {};
            admins.push({
              id: p.id,
              email: u.email || "?",
              permissions: p.permissions || [],
              is_superadmin: (u.email || "").toLowerCase() === SUPERADMIN_EMAIL,
            });
          }
          return json({ success: true, admins });
        }

        // --- Rattacher un rapport NinjaOne à un hôtel (admin) ---
        if (url.pathname === "/admin/rattacher-rapport") {
          const { rapportId, hotelId } = body;
          if (!rapportId || !hotelId) return json({ error: "rapportId et hotelId requis" }, 400);
          const upRes = await fetch(`${env.SUPABASE_URL}/rest/v1/ninjaone_rapports?id=eq.${rapportId}`, {
            method: "PATCH",
            headers: sbJsonHeaders(env),
            body: JSON.stringify({ hotel_id: hotelId, a_rattacher: false }),
          });
          if (!upRes.ok) {
            const e = await upRes.text().catch(() => "");
            return json({ error: "Erreur rattachement", detail: e.slice(0, 200) }, 500);
          }
          return json({ success: true });
        }

        // --- Définir les hôtels d'une procédure (remplace toutes les liaisons) ---
        if (url.pathname === "/admin/set-procedure-hotels") {
          const { procedureId, hotelIds } = body;
          if (!procedureId || !Array.isArray(hotelIds)) {
            return json({ error: "procedureId et hotelIds (tableau) requis" }, 400);
          }

          // 1. Supprime toutes les liaisons existantes de cette procédure
          const delRes = await fetch(`${env.SUPABASE_URL}/rest/v1/procedure_hotels?procedure_id=eq.${procedureId}`, {
            method: "DELETE",
            headers: sbHeaders(env),
          });
          if (!delRes.ok) {
            const e = await delRes.text().catch(() => "");
            return json({ error: "Erreur suppression des liaisons", detail: e.slice(0, 200) }, 500);
          }

          // 2. Réinsère une liaison par hôtel coché
          const uniqueIds = [...new Set(hotelIds.filter(Boolean))];
          if (uniqueIds.length > 0) {
            const rows = uniqueIds.map(hid => ({ procedure_id: procedureId, hotel_id: hid }));
            const insRes = await fetch(`${env.SUPABASE_URL}/rest/v1/procedure_hotels`, {
              method: "POST",
              headers: sbJsonHeaders(env),
              body: JSON.stringify(rows),
            });
            if (!insRes.ok) {
              const e = await insRes.text().catch(() => "");
              return json({ error: "Erreur insertion des liaisons", detail: e.slice(0, 200) }, 500);
            }

            // 3. Compatibilité : hotel_id d'origine = 1er hôtel coché
            await fetch(`${env.SUPABASE_URL}/rest/v1/procedures?id=eq.${procedureId}`, {
              method: "PATCH",
              headers: sbJsonHeaders(env),
              body: JSON.stringify({ hotel_id: uniqueIds[0] }),
            }).catch(() => {});
          }

          return json({ success: true, count: uniqueIds.length });
        }

        // --- Supprimer un rapport NinjaOne (ligne + fichier Storage) ---
        if (url.pathname === "/admin/supprimer-rapport") {
          const { rapportId } = body;
          if (!rapportId) return json({ error: "rapportId requis" }, 400);

          // Récupère l'URL du fichier pour en déduire le chemin Storage
          const getRes = await fetch(`${env.SUPABASE_URL}/rest/v1/ninjaone_rapports?id=eq.${rapportId}&select=file_url`, {
            headers: sbHeaders(env),
          });
          const rows = await getRes.json().catch(() => []);
          const fileUrl = rows[0] && rows[0].file_url ? rows[0].file_url : null;

          // Supprime le fichier du bucket "uploads" si on peut extraire le chemin
          if (fileUrl) {
            const marker = "/object/public/uploads/";
            const idx = fileUrl.indexOf(marker);
            if (idx !== -1) {
              const objectPath = fileUrl.slice(idx + marker.length);
              await fetch(`${env.SUPABASE_URL}/storage/v1/object/uploads/${objectPath}`, {
                method: "DELETE",
                headers: sbHeaders(env),
              }).catch(() => {});
            }
          }

          // Supprime la ligne
          const delRes = await fetch(`${env.SUPABASE_URL}/rest/v1/ninjaone_rapports?id=eq.${rapportId}`, {
            method: "DELETE",
            headers: sbHeaders(env),
          });
          if (!delRes.ok) {
            const e = await delRes.text().catch(() => "");
            return json({ error: "Erreur suppression", detail: e.slice(0, 200) }, 500);
          }
          return json({ success: true });
        }

        // --- Journal de connexions (super-admin uniquement) ---
        if (url.pathname === "/admin/login-logs") {
          if (!(await verifySuperAdmin(request, env))) return json({ error: "Réservé au super-admin" }, 403);

          const { limit, email } = body || {};
          const lim = Math.min(Math.max(parseInt(limit || 200, 10) || 200, 1), 1000);
          let query = `login_logs?select=id,email,role,logged_at&order=email.asc&limit=${lim}`;
          if (email && String(email).trim()) {
            query += `&email=ilike.*${encodeURIComponent(String(email).trim())}*`;
          }

          const res = await fetch(`${env.SUPABASE_URL}/rest/v1/${query}`, {
            headers: sbHeaders(env),
          });
          if (!res.ok) {
            const e = await res.text().catch(() => "");
            return json({ error: "Erreur lecture journal", detail: e.slice(0, 300) }, 500);
          }
          const logs = await res.json().catch(() => []);
          return json({ success: true, logs });
        }

        // --- Créer un admin (super-admin uniquement) + email de bienvenue ---
        if (url.pathname === "/admin/create-admin") {
          if (!(await verifySuperAdmin(request, env))) return json({ error: "Réservé au super-admin" }, 403);

          const { email, password, permissions, fullName } = body;
          if (!email || !password) return json({ error: "Email et mot de passe requis" }, 400);

          const uRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users`, {
            method: "POST",
            headers: sbJsonHeaders(env),
            body: JSON.stringify({ email, password, email_confirm: true }),
          });
          const uData = await uRes.json();
          if (!uRes.ok) return json({ error: "Erreur création compte", details: uData }, uRes.status);

          const profRes = await fetch(`${env.SUPABASE_URL}/rest/v1/profiles`, {
            method: "POST",
            headers: sbJsonHeaders(env),
            body: JSON.stringify({ id: uData.id, role: "admin", hotel_id: null, permissions: permissions || [], full_name: fullName || null }),
          });
          if (!profRes.ok) {
            await deleteAuthUserByEmail(env, email);
            const e = await profRes.json().catch(() => ({}));
            return json({ error: "Erreur création profil", details: e }, 500);
          }

          // Email de bienvenue avec les accès (n'échoue pas la création si l'envoi rate)
          await envoyerBienvenueAdmin(env, email, password, permissions || []);

          return json({ success: true, admin: { id: uData.id, email, permissions: permissions || [] } });
        }

        // --- Modifier les permissions (super-admin uniquement) ---
        if (url.pathname === "/admin/update-admin-permissions") {
          if (!(await verifySuperAdmin(request, env))) return json({ error: "Réservé au super-admin" }, 403);

          const { adminId, permissions } = body;
          if (!adminId) return json({ error: "adminId requis" }, 400);

          const upRes = await fetch(`${env.SUPABASE_URL}/rest/v1/profiles?id=eq.${adminId}`, {
            method: "PATCH",
            headers: sbJsonHeaders(env),
            body: JSON.stringify({ permissions: permissions || [] }),
          });
          if (!upRes.ok) return json({ error: "Erreur mise à jour" }, 500);
          return json({ success: true });
        }

        // --- Téléphones MFA d'un administrateur (super-admin uniquement) ---
        // Supabase ne permet pas d'enrôler un téléphone pour un autre compte : on liste,
        // retire un téléphone, ou réinitialise la MFA. À sa prochaine connexion, l'admin
        // concerné voit automatiquement un nouveau QR code à scanner.
        if (["/admin/list-admin-factors", "/admin/delete-admin-factor", "/admin/reset-admin-mfa"].includes(url.pathname)) {
          const sup = await verifySuperAdmin(request, env);
          if (!sup) return json({ error: "Réservé au super-admin" }, 403);
          // Exige une session validée par la double authentification (AAL2)
          let aal = "";
          try {
            const tk = (request.headers.get("Authorization") || "").replace("Bearer ", "");
            const part = tk.split(".")[1].replace(/-/g, "+").replace(/_/g, "/");
            aal = JSON.parse(atob(part + "=".repeat((4 - part.length % 4) % 4))).aal || "";
          } catch (e) {}
          if (aal !== "aal2") return json({ error: "Double authentification requise pour cette action." }, 403);

          const adminId = String(body.adminId || "");
          if (!/^[0-9a-f-]{36}$/i.test(adminId)) return json({ error: "adminId invalide" }, 400);
          if (adminId === sup.id) return json({ error: "Gérez vos propres téléphones depuis la page « Double authentification »." }, 400);
          const pRes = await fetch(`${env.SUPABASE_URL}/rest/v1/profiles?id=eq.${adminId}&select=role`, { headers: sbHeaders(env) });
          const prof = pRes.ok ? await pRes.json() : [];
          if (!prof[0] || prof[0].role !== "admin") return json({ error: "Ce compte n'est pas un administrateur" }, 400);

          const listerFacteurs = async () => {
            const r = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users/${adminId}`, { headers: sbHeaders(env) });
            if (!r.ok) throw new Error("Lecture du compte impossible (" + r.status + ")");
            const u = await r.json();
            return (u.factors || []).map(f => ({ id: f.id, friendly_name: f.friendly_name || "", factor_type: f.factor_type, status: f.status, created_at: f.created_at }));
          };
          const supprimerFacteur = async (factorId) => {
            const r = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users/${adminId}/factors/${factorId}`, { method: "DELETE", headers: sbHeaders(env) });
            if (!r.ok && r.status !== 404) throw new Error("Suppression refusée (" + r.status + ") : " + (await r.text()).slice(0, 200));
          };

          if (url.pathname === "/admin/list-admin-factors") {
            return json({ success: true, factors: await listerFacteurs() });
          }
          if (url.pathname === "/admin/delete-admin-factor") {
            const factorId = String(body.factorId || "");
            const facteurs = await listerFacteurs();
            if (!facteurs.some(f => f.id === factorId)) return json({ error: "Téléphone introuvable pour ce compte" }, 404);
            await supprimerFacteur(factorId);
            return json({ success: true });
          }
          // /admin/reset-admin-mfa : retire tous les téléphones
          const facteurs = await listerFacteurs();
          for (const f of facteurs) await supprimerFacteur(f.id);
          return json({ success: true, supprimes: facteurs.length });
        }

        // --- Supprimer un admin (super-admin uniquement) ---
        if (url.pathname === "/admin/delete-admin") {
          if (!(await verifySuperAdmin(request, env))) return json({ error: "Réservé au super-admin" }, 403);

          const { adminId, adminEmail } = body;
          if (adminEmail && adminEmail.toLowerCase() === SUPERADMIN_EMAIL) {
            return json({ error: "Impossible de supprimer le super-admin" }, 400);
          }
          if (adminEmail) await deleteAuthUserByEmail(env, adminEmail);
          if (adminId) {
            await fetch(`${env.SUPABASE_URL}/rest/v1/profiles?id=eq.${adminId}`, {
              method: "DELETE", headers: sbHeaders(env),
            });
          }
          return json({ success: true });
        }

        return json({ error: "Route admin inconnue" }, 404);
      } catch (err) {
        return json({ error: "Erreur serveur inattendue", message: err.message }, 500);
      }
    }

    return json({ error: "Route inconnue" }, 404);
  },
};

// Point d'entrée : nettoie les variables (espaces / retours à la ligne collés par erreur
// dans Cloudflare, qui font planter fetch() avec "Invalid header value") et garantit
// une réponse JSON avec en-têtes CORS même en cas d'erreur inattendue.
export default {
  async fetch(request, env, ctx) {
    const envPropre = { ...env };
    for (const k of ["SUPABASE_URL", "SUPABASE_SERVICE_ROLE_KEY", "MS_CLIENT_ID", "MS_CLIENT_SECRET", "MS_TENANT_ID"]) {
      if (typeof envPropre[k] === "string") envPropre[k] = envPropre[k].trim();
    }
    try {
      return await app.fetch(request, envPropre, ctx);
    } catch (e) {
      CURRENT_ORIGIN = request.headers.get("Origin") || ALLOWED_ORIGINS[0];
      return json({ error: "Erreur interne du Worker : " + String((e && e.message) || e) }, 500);
    }
  },
};
